```sql
-- FILE: supabase/migrations/000_extensions.sql
-- PURPOSE: Enable PostgreSQL extensions required by Atlas.
--   uuid-ossp  → deterministic UUID generation for seed data
--   pgcrypto   → AES encryption for OAuth tokens at rest
--   pg_trgm    → trigram similarity for subject/chapter search
--   unaccent   → accent-insensitive text search

CREATE EXTENSION IF NOT EXISTS "pgcrypto";
CREATE EXTENSION IF NOT EXISTS "pg_trgm";
CREATE EXTENSION IF NOT EXISTS "unaccent";
```

```sql
-- ============================================================
-- FILE: supabase/migrations/001_enums.sql
-- PURPOSE: All custom ENUM types used across Atlas tables.
-- ENUMs enforce valid states at the DB layer with zero app code.
-- Adding new values: ALTER TYPE x ADD VALUE 'new_val';
-- ============================================================

-- Notes completion state for a chapter
CREATE TYPE notes_status_enum AS ENUM (
  'none',          -- Student has not started notes
  'in_progress',   -- Student is working on notes
  'complete'       -- Notes are finished
);

-- Mission types generated by the Mission Engine
CREATE TYPE mission_type_enum AS ENUM (
  'complete_notes',      -- Write/finish chapter notes
  'review_chapter',      -- Revisit a previously completed chapter
  'attempt_paper',       -- Do a full past paper
  'revisit_weak_topic',  -- Re-study a low-accuracy chapter
  'confidence_check'     -- Update self-assessment for a chapter
);

-- Lifecycle state of a daily mission
CREATE TYPE mission_status_enum AS ENUM (
  'pending',    -- Not yet acted on
  'completed',  -- Student marked as done
  'skipped'     -- Student chose to skip
);

-- CAIE exam session identifiers
CREATE TYPE paper_session_enum AS ENUM (
  'feb_mar',  -- February/March series (limited subjects)
  'may_jun',  -- May/June series (main sitting)
  'oct_nov'   -- October/November series (main sitting)
);

-- All sources of XP in the system (immutable event ledger)
CREATE TYPE xp_event_type_enum AS ENUM (
  'mission_complete',    -- Completed a daily mission (+bonus)
  'notes_complete',      -- Marked chapter notes as done
  'notes_in_progress',   -- Started chapter notes
  'paper_attempt',       -- Logged a past paper attempt
  'streak_bonus_7',      -- 7-day streak milestone
  'streak_bonus_30',     -- 30-day streak milestone
  'streak_bonus_100',    -- 100-day streak milestone
  'achievement_unlock',  -- Earned a badge
  'confidence_update',   -- Updated confidence self-rating
  'friend_challenge_win',-- Won a PvP study challenge (future)
  'manual_adjustment'    -- Admin or correction (future)
);

-- Social relationship state (future friends feature)
CREATE TYPE friendship_status_enum AS ENUM (
  'pending',   -- Request sent, not yet accepted
  'accepted',  -- Mutual friendship established
  'blocked'    -- One user blocked the other
);

-- PvP challenge lifecycle (future)
CREATE TYPE challenge_status_enum AS ENUM (
  'pending',    -- Waiting for opponent to accept
  'active',     -- Challenge is live
  'completed',  -- One player won
  'cancelled',  -- Abandoned or declined
  'expired'     -- Time limit exceeded
);

-- Polymorphic entity type reference (missions, challenges)
CREATE TYPE entity_type_enum AS ENUM (
  'chapter',
  'subject',
  'paper',
  'user'
);

-- Notification delivery types
CREATE TYPE notification_type_enum AS ENUM (
  'mission_reminder',    -- Daily study reminder
  'streak_warning',      -- Streak about to break
  'achievement_unlock',  -- Badge earned
  'friend_request',      -- Social: incoming request
  'challenge_invite',    -- PvP: challenge invitation
  'challenge_result',    -- PvP: challenge ended
  'level_up',            -- Levelled up
  'system'               -- Atlas system message
);

-- Coin transaction types (future shop)
CREATE TYPE coin_transaction_type_enum AS ENUM (
  'earned_mission',
  'earned_streak',
  'earned_achievement',
  'spent_shop',
  'spent_challenge',
  'admin_adjustment'
);
```

```sql
-- ============================================================
-- FILE: supabase/migrations/002_core_tables.sql
-- PURPOSE: The foundational data model.
--
--   profiles         → Extends auth.users. One row per student.
--   subjects         → Global CAIE subject catalog (seeded).
--   user_subjects    → Which subjects a student is enrolled in.
--   chapters         → Global syllabus chapters per subject (seeded).
--   user_chapters    → Per-student progress state per chapter.
--   past_papers      → Student's paper attempt log.
--   paper_question_attempts → Granular per-question breakdown.
--
-- All user-owned tables carry user_id as the RLS anchor.
-- ============================================================

-- ─── PROFILES ─────────────────────────────────────────────────────────────
-- Mirrors auth.users. Created automatically via trigger on signup.
-- total_xp and current_level are denormalised here for fast reads
-- (avoiding expensive aggregation on every dashboard load).
-- The xp_events table is the canonical ledger; this is the cache.

CREATE TABLE public.profiles (
  id                    UUID        PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
  email                 TEXT        NOT NULL,
  full_name             TEXT,
  avatar_url            TEXT,
  school                TEXT,
  exam_session          TEXT,                          -- e.g. "Jun 2026"
  timezone              TEXT        NOT NULL DEFAULT 'UTC',
  onboarding_completed  BOOLEAN     NOT NULL DEFAULT FALSE,
  total_xp              INTEGER     NOT NULL DEFAULT 0 CHECK (total_xp >= 0),
  current_level         SMALLINT    NOT NULL DEFAULT 1 CHECK (current_level BETWEEN 1 AND 100),
  total_coins           INTEGER     NOT NULL DEFAULT 0 CHECK (total_coins >= 0), -- future shop
  created_at            TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at            TIMESTAMPTZ NOT NULL DEFAULT NOW(),

  CONSTRAINT profiles_email_check CHECK (email ~* '^[^@]+@[^@]+\.[^@]+$')
);

COMMENT ON TABLE public.profiles IS
  'Extended user profile. Extends auth.users. total_xp is a denormalised sum of xp_events.';
COMMENT ON COLUMN public.profiles.total_xp IS
  'Denormalised sum kept in sync by the sync_xp_to_profile trigger. Source of truth is xp_events.';
COMMENT ON COLUMN public.profiles.total_coins IS
  'Stub column for future coin economy. Always 0 until shop feature launches.';


-- ─── SUBJECTS ─────────────────────────────────────────────────────────────
-- Global catalog of CAIE A-Level subjects. Seeded at deploy time.
-- is_global = true means it's part of the official CAIE catalog.
-- is_global = false means a student created a custom subject.
-- created_by is null for global subjects.

CREATE TABLE public.subjects (
  id           UUID        PRIMARY KEY DEFAULT uuid_generate_v4(),
  name         TEXT        NOT NULL,
  code         TEXT,                                   -- e.g. "9709"
  color_hex    TEXT        NOT NULL DEFAULT '#5B7FFF', -- UI theme color
  icon         TEXT        NOT NULL DEFAULT 'BookOpen', -- Lucide icon name
  is_global    BOOLEAN     NOT NULL DEFAULT TRUE,
  created_by   UUID        REFERENCES public.profiles(id) ON DELETE SET NULL,
  created_at   TIMESTAMPTZ NOT NULL DEFAULT NOW(),

  CONSTRAINT subjects_name_not_empty CHECK (LENGTH(TRIM(name)) > 0),
  CONSTRAINT subjects_color_hex_format CHECK (color_hex ~* '^#[0-9A-Fa-f]{6}$')
);

COMMENT ON TABLE public.subjects IS
  'CAIE A-Level subject catalog. Global subjects seeded; students can add custom ones.';


-- ─── USER_SUBJECTS ─────────────────────────────────────────────────────────
-- Junction: subjects a student is actively studying.
-- exam_date and target_grade drive the Mission Engine urgency calculations.

CREATE TABLE public.user_subjects (
  id             UUID        PRIMARY KEY DEFAULT uuid_generate_v4(),
  user_id        UUID        NOT NULL REFERENCES public.profiles(id) ON DELETE CASCADE,
  subject_id     UUID        NOT NULL REFERENCES public.subjects(id) ON DELETE CASCADE,
  exam_date      DATE,
  target_grade   TEXT        CHECK (target_grade IN ('A*', 'A', 'B', 'C', 'D', 'E')),
  priority       SMALLINT    NOT NULL DEFAULT 3 CHECK (priority BETWEEN 1 AND 5),
  is_archived    BOOLEAN     NOT NULL DEFAULT FALSE,  -- soft-delete for "finished" subjects
  created_at     TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at     TIMESTAMPTZ NOT NULL DEFAULT NOW(),

  CONSTRAINT user_subjects_unique UNIQUE (user_id, subject_id)
);

COMMENT ON TABLE public.user_subjects IS
  'Subjects a student is enrolled in. Priority (1-5) weights the Mission Engine scoring.';
COMMENT ON COLUMN public.user_subjects.is_archived IS
  'Set true after exam date passes. Archived subjects are hidden but data is preserved.';


-- ─── CHAPTERS ─────────────────────────────────────────────────────────────
-- Global syllabus chapters per subject. Seeded from official CAIE syllabuses.
-- Students cannot create chapters for global subjects (only admins can).
-- For custom subjects (is_global=false), students can add their own chapters.

CREATE TABLE public.chapters (
  id           UUID        PRIMARY KEY DEFAULT uuid_generate_v4(),
  subject_id   UUID        NOT NULL REFERENCES public.subjects(id) ON DELETE CASCADE,
  title        TEXT        NOT NULL,
  number       SMALLINT    NOT NULL DEFAULT 1 CHECK (number > 0),
  component    TEXT,                                   -- e.g. "Pure 1", "Statistics 1", "Mechanics"
  description  TEXT,                                   -- brief syllabus summary
  is_global    BOOLEAN     NOT NULL DEFAULT TRUE,
  created_at   TIMESTAMPTZ NOT NULL DEFAULT NOW(),

  CONSTRAINT chapters_title_not_empty CHECK (LENGTH(TRIM(title)) > 0),
  CONSTRAINT chapters_unique_number_per_component UNIQUE (subject_id, component, number)
);

COMMENT ON TABLE public.chapters IS
  'Syllabus chapters. Global chapters seeded from CAIE syllabuses. component groups chapters by paper.';


-- ─── USER_CHAPTERS ─────────────────────────────────────────────────────────
-- Per-student progress state for each chapter they are tracking.
-- This is the primary input to the Mission Engine algorithm.
-- Created lazily: a row is created when a student first interacts with a chapter.

CREATE TABLE public.user_chapters (
  id                UUID             PRIMARY KEY DEFAULT uuid_generate_v4(),
  user_id           UUID             NOT NULL REFERENCES public.profiles(id) ON DELETE CASCADE,
  chapter_id        UUID             NOT NULL REFERENCES public.chapters(id) ON DELETE CASCADE,
  notes_status      notes_status_enum NOT NULL DEFAULT 'none',
  google_doc_url    TEXT,
  google_doc_id     TEXT,                              -- for Drive API sync
  confidence_level  SMALLINT         CHECK (confidence_level BETWEEN 1 AND 5),
  last_reviewed_at  TIMESTAMPTZ,                       -- updated when student reviews or completes
  personal_notes    TEXT,                              -- freeform text notes (short)
  created_at        TIMESTAMPTZ      NOT NULL DEFAULT NOW(),
  updated_at        TIMESTAMPTZ      NOT NULL DEFAULT NOW(),

  CONSTRAINT user_chapters_unique UNIQUE (user_id, chapter_id),
  CONSTRAINT user_chapters_google_doc_url_format
    CHECK (google_doc_url IS NULL OR google_doc_url ~* '^https?://')
);

COMMENT ON TABLE public.user_chapters IS
  'Per-student chapter progress. notes_status and confidence_level are key Mission Engine inputs.';
COMMENT ON COLUMN public.user_chapters.last_reviewed_at IS
  'Drives the recency_penalty in the Mission Engine. Chapters not reviewed recently score higher.';


-- ─── PAST_PAPERS ─────────────────────────────────────────────────────────
-- Each row is one completed paper attempt by a student.
-- accuracy_pct is a generated column (no trigger needed).
-- This feeds paper accuracy into the Readiness Score formula.

CREATE TABLE public.past_papers (
  id               UUID              PRIMARY KEY DEFAULT uuid_generate_v4(),
  user_id          UUID              NOT NULL REFERENCES public.profiles(id) ON DELETE CASCADE,
  subject_id       UUID              NOT NULL REFERENCES public.subjects(id) ON DELETE CASCADE,
  paper_code       TEXT              NOT NULL,         -- e.g. "9709/12"
  year             SMALLINT          NOT NULL CHECK (year BETWEEN 1990 AND 2100),
  session          paper_session_enum NOT NULL,
  paper_number     SMALLINT          CHECK (paper_number BETWEEN 1 AND 9),
  attempted_at     DATE              NOT NULL DEFAULT CURRENT_DATE,
  score_raw        SMALLINT          NOT NULL CHECK (score_raw >= 0),
  score_max        SMALLINT          NOT NULL CHECK (score_max > 0),
  accuracy_pct     NUMERIC(5,2)      GENERATED ALWAYS AS (
                     ROUND((score_raw::NUMERIC / score_max) * 100, 2)
                   ) STORED,
  time_taken_mins  SMALLINT          CHECK (time_taken_mins > 0),
  notes            TEXT,
  created_at       TIMESTAMPTZ       NOT NULL DEFAULT NOW(),

  CONSTRAINT past_papers_score_valid CHECK (score_raw <= score_max),
  CONSTRAINT past_papers_unique_attempt
    UNIQUE (user_id, paper_code, year, session, attempted_at) -- prevents duplicate logging
);

COMMENT ON TABLE public.past_papers IS
  'Student past paper attempts. accuracy_pct is generated from score_raw/score_max. '
  'Feeds Readiness Score (40% weight) and Mission Engine (accuracy_gap signal).';
COMMENT ON COLUMN public.past_papers.accuracy_pct IS
  'GENERATED ALWAYS: (score_raw / score_max) * 100. Automatically consistent.';


-- ─── PAPER_QUESTION_ATTEMPTS ──────────────────────────────────────────────
-- Granular per-question breakdown of a paper attempt.
-- chapter_id tags which syllabus chapter a question tests.
-- This drives the per-chapter accuracy_gap in the Mission Engine.
-- Optional: students can log paper totals without this level of detail.

CREATE TABLE public.paper_question_attempts (
  id               UUID     PRIMARY KEY DEFAULT uuid_generate_v4(),
  paper_id         UUID     NOT NULL REFERENCES public.past_papers(id) ON DELETE CASCADE,
  chapter_id       UUID     REFERENCES public.chapters(id) ON DELETE SET NULL,
  question_number  TEXT     NOT NULL,                  -- e.g. "5", "5(b)", "Q3a"
  marks_available  SMALLINT NOT NULL CHECK (marks_available > 0),
  marks_obtained   SMALLINT NOT NULL CHECK (marks_obtained >= 0),
  created_at       TIMESTAMPTZ NOT NULL DEFAULT NOW(),

  CONSTRAINT paper_question_marks_valid CHECK (marks_obtained <= marks_available)
);

COMMENT ON TABLE public.paper_question_attempts IS
  'Granular per-question data. chapter_id enables per-chapter accuracy tracking '
  'which is the primary weak-topic detection signal in the Mission Engine.';
```

```sql
-- ============================================================
-- FILE: supabase/migrations/003_gamification_tables.sql
-- PURPOSE: The gamification core — missions, XP, streaks, badges.
--
--   achievement_definitions → Catalog of all possible badges.
--   daily_missions          → Generated missions per user per day.
--   xp_events               → Immutable ledger of all XP earned.
--   streaks                 → Daily study streak tracking.
--   user_achievements       → Which badges a user has unlocked.
-- ============================================================

-- ─── ACHIEVEMENT_DEFINITIONS ──────────────────────────────────────────────
-- Central catalog of all achievable badges. Stored in DB (not just code)
-- so achievements can be added/modified without a code deploy.
-- Conditions are evaluated server-side in check_and_unlock_achievements().

CREATE TABLE public.achievement_definitions (
  key              TEXT        PRIMARY KEY,             -- e.g. "streak_7", "first_blood"
  name             TEXT        NOT NULL,               -- Display name
  description      TEXT        NOT NULL,               -- How to earn it
  icon             TEXT        NOT NULL DEFAULT 'Trophy', -- Lucide icon
  color_hex        TEXT        NOT NULL DEFAULT '#FFD166',
  xp_reward        SMALLINT    NOT NULL DEFAULT 100,
  is_active        BOOLEAN     NOT NULL DEFAULT TRUE,  -- disable without deleting
  is_secret        BOOLEAN     NOT NULL DEFAULT FALSE, -- hidden until unlocked
  sort_order       SMALLINT    NOT NULL DEFAULT 99,
  created_at       TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

COMMENT ON TABLE public.achievement_definitions IS
  'Badge catalog. Stored in DB for runtime flexibility. '
  'check_and_unlock_achievements() evaluates conditions against this table.';
COMMENT ON COLUMN public.achievement_definitions.is_secret IS
  'Secret achievements are shown as "???" in UI until unlocked.';


-- ─── DAILY_MISSIONS ───────────────────────────────────────────────────────
-- Generated by generate_daily_missions() each day per user.
-- Max 3 missions per user per day (enforced by UNIQUE + app logic).
-- The UNIQUE constraint prevents duplicate mission generation if called twice.
-- target_entity_id is polymorphic: points to user_chapters.id or subjects.id.

CREATE TABLE public.daily_missions (
  id                  UUID              PRIMARY KEY DEFAULT uuid_generate_v4(),
  user_id             UUID              NOT NULL REFERENCES public.profiles(id) ON DELETE CASCADE,
  mission_date        DATE              NOT NULL DEFAULT CURRENT_DATE,
  type                mission_type_enum NOT NULL,
  target_entity_type  entity_type_enum  NOT NULL DEFAULT 'chapter',
  target_entity_id    UUID,                             -- polymorphic FK
  title               TEXT              NOT NULL,
  description         TEXT,
  xp_reward           SMALLINT          NOT NULL DEFAULT 30 CHECK (xp_reward >= 0),
  status              mission_status_enum NOT NULL DEFAULT 'pending',
  completed_at        TIMESTAMPTZ,
  generated_at        TIMESTAMPTZ       NOT NULL DEFAULT NOW(),

  -- Prevents the same mission being generated twice for same day
  CONSTRAINT daily_missions_unique
    UNIQUE (user_id, mission_date, type, target_entity_id),

  -- completed_at must only be set when status is 'completed'
  CONSTRAINT daily_missions_completed_at_check
    CHECK (
      (status = 'completed' AND completed_at IS NOT NULL) OR
      (status != 'completed' AND completed_at IS NULL)
    )
);

COMMENT ON TABLE public.daily_missions IS
  'Daily missions generated by the Mission Engine algorithm. '
  'UNIQUE constraint prevents duplicate generation. Max 3 per user per day enforced in function.';
COMMENT ON COLUMN public.daily_missions.target_entity_id IS
  'Polymorphic. Points to user_chapters.id (chapter missions) or subjects.id (paper missions). '
  'Interpret based on target_entity_type.';


-- ─── XP_EVENTS ────────────────────────────────────────────────────────────
-- Immutable append-only ledger of every XP transaction.
-- NEVER delete or update rows here — integrity of total_xp depends on it.
-- profiles.total_xp is a denormalised sum maintained by trigger.
-- xp_amount can be negative in future (penalties, corrections).

CREATE TABLE public.xp_events (
  id            UUID                PRIMARY KEY DEFAULT uuid_generate_v4(),
  user_id       UUID                NOT NULL REFERENCES public.profiles(id) ON DELETE CASCADE,
  event_type    xp_event_type_enum  NOT NULL,
  xp_amount     SMALLINT            NOT NULL,  -- can be negative for corrections
  reference_id  UUID,                          -- polymorphic: mission_id, paper_id, etc.
  metadata      JSONB               NOT NULL DEFAULT '{}',
  created_at    TIMESTAMPTZ         NOT NULL DEFAULT NOW()
);

COMMENT ON TABLE public.xp_events IS
  'Immutable XP ledger. NEVER UPDATE OR DELETE. profiles.total_xp is synced via trigger. '
  'Sum of xp_events.xp_amount per user = profiles.total_xp.';


-- ─── STREAKS ──────────────────────────────────────────────────────────────
-- One row per user. Updated by update_streak() on any study activity.
-- last_activity_date is compared in user's LOCAL timezone (from profiles.timezone).
-- A streak is "active" if last_activity_date = today (user TZ) or yesterday.
-- Streak freezes (future feature): freeze_count allows future "streak shield" items.

CREATE TABLE public.streaks (
  id                  UUID        PRIMARY KEY DEFAULT uuid_generate_v4(),
  user_id             UUID        NOT NULL REFERENCES public.profiles(id) ON DELETE CASCADE,
  current_streak      INTEGER     NOT NULL DEFAULT 0 CHECK (current_streak >= 0),
  longest_streak      INTEGER     NOT NULL DEFAULT 0 CHECK (longest_streak >= 0),
  last_activity_date  DATE,
  freeze_count        SMALLINT    NOT NULL DEFAULT 0 CHECK (freeze_count >= 0), -- future streak shields
  updated_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),

  CONSTRAINT streaks_user_unique UNIQUE (user_id),
  CONSTRAINT streaks_longest_gte_current
    CHECK (longest_streak >= current_streak)
);

COMMENT ON TABLE public.streaks IS
  'Streak tracking. update_streak() is called on every study activity. '
  'Uses profiles.timezone for correct midnight boundary calculation. '
  'freeze_count is a stub for future "streak shield" shop items.';


-- ─── USER_ACHIEVEMENTS ────────────────────────────────────────────────────
-- Append-only record of which badges a student has earned.
-- check_and_unlock_achievements() writes here after evaluating conditions.
-- The UNIQUE constraint is critical: prevents duplicate badge grants.

CREATE TABLE public.user_achievements (
  id               UUID        PRIMARY KEY DEFAULT uuid_generate_v4(),
  user_id          UUID        NOT NULL REFERENCES public.profiles(id) ON DELETE CASCADE,
  achievement_key  TEXT        NOT NULL REFERENCES public.achievement_definitions(key) ON DELETE RESTRICT,
  unlocked_at      TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  notified         BOOLEAN     NOT NULL DEFAULT FALSE, -- track if UI notification was shown

  CONSTRAINT user_achievements_unique UNIQUE (user_id, achievement_key)
);

COMMENT ON TABLE public.user_achievements IS
  'Unlocked badges. Append-only: never delete rows. '
  'UNIQUE prevents duplicate grants even if check_and_unlock_achievements() is called concurrently.';
```

```sql
-- FILE: supabase/migrations/004_integration_tables.sql
-- PURPOSE: External integrations, notifications, and user settings.

--   google_docs_tokens → Encrypted OAuth2 tokens for Google Docs.
--   notifications      → In-app notification queue.
--   user_settings      → Per-user preferences (notification prefs, etc.)

-- ─── GOOGLE_DOCS_TOKENS ───────────────────────────────────────────────────
-- Stores Google OAuth2 tokens encrypted with pgcrypto.
-- In production: use Supabase Vault for the encryption key.
-- access_token and refresh_token are AES-256 encrypted strings.
-- Never expose via RLS to the client — server-only reads via SECURITY DEFINER functions.

CREATE TABLE public.google_docs_tokens (
  id             UUID        PRIMARY KEY DEFAULT uuid_generate_v4(),
  user_id        UUID        NOT NULL REFERENCES public.profiles(id) ON DELETE CASCADE,
  -- Tokens stored encrypted: pgp_sym_encrypt(token, key) on write
  -- Decrypted: pgp_sym_decrypt(access_token_enc::bytea, key) on read
  access_token_enc   TEXT    NOT NULL,  -- pgcrypto encrypted
  refresh_token_enc  TEXT    NOT NULL,  -- pgcrypto encrypted
  expires_at     TIMESTAMPTZ NOT NULL,
  scopes         TEXT[]      NOT NULL DEFAULT '{}',
  google_user_id TEXT,                  -- Google's user sub identifier
  google_email   TEXT,                  -- Which Google account is connected
  created_at     TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at     TIMESTAMPTZ NOT NULL DEFAULT NOW(),

  CONSTRAINT google_docs_tokens_user_unique UNIQUE (user_id)
);

COMMENT ON TABLE public.google_docs_tokens IS
  'Google OAuth2 tokens encrypted with pgcrypto. '
  'RLS blocks ALL client access — tokens are only readable by SECURITY DEFINER server functions. '
  'Production: migrate to Supabase Vault for key management.';


-- ─── NOTIFICATIONS ────────────────────────────────────────────────────────
-- Server-generated notification queue. Consumed and dismissed by the client.
-- Supports future push notification delivery tracking.

CREATE TABLE public.notifications (
  id             UUID                     PRIMARY KEY DEFAULT uuid_generate_v4(),
  user_id        UUID                     NOT NULL REFERENCES public.profiles(id) ON DELETE CASCADE,
  type           notification_type_enum   NOT NULL,
  title          TEXT                     NOT NULL,
  body           TEXT,
  data           JSONB                    NOT NULL DEFAULT '{}', -- action payload
  is_read        BOOLEAN                  NOT NULL DEFAULT FALSE,
  is_sent_push   BOOLEAN                  NOT NULL DEFAULT FALSE,
  created_at     TIMESTAMPTZ              NOT NULL DEFAULT NOW(),
  read_at        TIMESTAMPTZ
);

COMMENT ON TABLE public.notifications IS
  'In-app notification queue. is_read tracks dismissal. '
  'is_sent_push tracks push notification delivery (future). '
  'data JSONB carries action payloads (e.g. {achievement_key: "streak_7"}).';


-- ─── USER_SETTINGS ────────────────────────────────────────────────────────
-- Per-user preferences. Separate from profiles to keep profiles lean.
-- All columns have sensible defaults so a row is created once and updated.

CREATE TABLE public.user_settings (
  id                       UUID        PRIMARY KEY DEFAULT uuid_generate_v4(),
  user_id                  UUID        NOT NULL REFERENCES public.profiles(id) ON DELETE CASCADE,
  -- Notification preferences
  notif_mission_reminder   BOOLEAN     NOT NULL DEFAULT TRUE,
  notif_streak_warning     BOOLEAN     NOT NULL DEFAULT TRUE,
  notif_achievement        BOOLEAN     NOT NULL DEFAULT TRUE,
  notif_friend_request     BOOLEAN     NOT NULL DEFAULT TRUE,
  notif_challenge          BOOLEAN     NOT NULL DEFAULT TRUE,
  -- Study reminder time (stored as HH:MM in user's timezone)
  reminder_time            TIME,
  -- UI preferences
  sidebar_collapsed        BOOLEAN     NOT NULL DEFAULT FALSE,
  theme                    TEXT        NOT NULL DEFAULT 'dark',  -- future themes
  -- Mission Engine preferences
  max_missions_per_day     SMALLINT    NOT NULL DEFAULT 3 CHECK (max_missions_per_day BETWEEN 1 AND 5),
  updated_at               TIMESTAMPTZ NOT NULL DEFAULT NOW(),

  CONSTRAINT user_settings_user_unique UNIQUE (user_id)
);

COMMENT ON TABLE public.user_settings IS
  'Per-user preferences. theme is a stub for the future themes shop feature. '
  'max_missions_per_day lets power users get up to 5 missions.';
```

```sql
-- ============================================================
-- FILE: supabase/migrations/005_future_tables.sql
-- PURPOSE: Stub tables for upcoming features.
-- These are created now so foreign keys and indexes can be
-- established from day one, avoiding painful migrations later.
-- All future tables have is_active guards or status ENUMs
-- so they can be ignored by the application until launched.
--

-- ─── FRIENDSHIPS ──────────────────────────────────────────────────────────
-- Stores accepted mutual friendships.
-- ENFORCED: user_id_1 < user_id_2 (lexicographic) to prevent duplicates.
-- This means "Alex ↔ Priya" is one row, not two.

CREATE TABLE public.friendships (
  id          UUID                   PRIMARY KEY DEFAULT uuid_generate_v4(),
  user_id_1   UUID                   NOT NULL REFERENCES public.profiles(id) ON DELETE CASCADE,
  user_id_2   UUID                   NOT NULL REFERENCES public.profiles(id) ON DELETE CASCADE,
  status      friendship_status_enum NOT NULL DEFAULT 'accepted',
  created_at  TIMESTAMPTZ            NOT NULL DEFAULT NOW(),

  CONSTRAINT friendships_no_self_friendship CHECK (user_id_1 != user_id_2),
  CONSTRAINT friendships_ordered CHECK (user_id_1 < user_id_2),
  CONSTRAINT friendships_unique UNIQUE (user_id_1, user_id_2)
);

COMMENT ON TABLE public.friendships IS
  'Mutual friend relationships. user_id_1 < user_id_2 prevents duplicate rows. '
  'Query: WHERE user_id_1 = $uid OR user_id_2 = $uid to find all friends of a user.';

COMMENT ON TABLE public.friend_requests IS
```

```sql
-- ─── UPDATE_STREAK ─────────────────────────────────────────────────────────
-- Called whenever any study activity occurs (mission complete, notes update, paper log).
-- Computes "today" in the user's local timezone for correct midnight boundary.
-- Uses FOR UPDATE lock to prevent concurrent race conditions.

CREATE OR REPLACE FUNCTION public.update_streak(p_user_id UUID)
RETURNS VOID
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_streak  public.streaks%ROWTYPE;
  v_today   DATE;
  v_user_tz TEXT;
BEGIN
  -- Get user's timezone for correct midnight boundary
  SELECT timezone INTO v_user_tz
  FROM public.profiles
  WHERE id = p_user_id;

  v_today := (NOW() AT TIME ZONE COALESCE(v_user_tz, 'UTC'))::DATE;

  -- Lock the row to prevent race conditions (e.g. two simultaneous mission completions)
  SELECT * INTO v_streak
  FROM public.streaks
  WHERE user_id = p_user_id
  FOR UPDATE;

  IF NOT FOUND THEN
    -- Safety fallback: create streak if missing
    INSERT INTO public.streaks (user_id, current_streak, longest_streak, last_activity_date)
    VALUES (p_user_id, 1, 1, v_today);
    RETURN;
  END IF;

  -- Already logged activity today — no change needed
  IF v_streak.last_activity_date = v_today THEN
    RETURN;
  END IF;

  IF v_streak.last_activity_date = v_today - INTERVAL '1 day' THEN
    -- Consecutive day: increment streak
    UPDATE public.streaks
    SET
      current_streak     = current_streak + 1,
      longest_streak     = GREATEST(longest_streak, current_streak + 1),
      last_activity_date = v_today,
      updated_at         = NOW()
    WHERE user_id = p_user_id;

    -- Award streak milestone XP
    IF (v_streak.current_streak + 1) = 7 THEN
      PERFORM public.award_xp(p_user_id, 150, 'streak_bonus_7',
        NULL, '{"streak_days": 7}'::JSONB);
    ELSIF (v_streak.current_streak + 1) = 30 THEN
      PERFORM public.award_xp(p_user_id, 500, 'streak_bonus_30',
        NULL, '{"streak_days": 30}'::JSONB);
    ELSIF (v_streak.current_streak + 1) = 100 THEN
      PERFORM public.award_xp(p_user_id, 2000, 'streak_bonus_100',
        NULL, '{"streak_days": 100}'::JSONB);
    END IF;

  ELSE
    -- Streak broken: reset to 1
    UPDATE public.streaks
    SET
      current_streak     = 1,
      last_activity_date = v_today,
      updated_at         = NOW()
    WHERE user_id = p_user_id;
  END IF;

END;
$$;

COMMENT ON FUNCTION public.update_streak IS
  'Call on any study activity. Handles consecutive-day detection in user local timezone. '
  'Grants streak milestone XP at 7, 30, and 100 days. Uses FOR UPDATE for race safety.';


-- ─── GENERATE_DAILY_MISSIONS ───────────────────────────────────────────────
-- The Mission Engine. Called on login if no missions exist for today.
-- Algorithm (per the system design):
--   For each chapter in enrolled subjects with future exam dates:
--     chapter_score = (
--       notes_gap     × 0.25
--       confidence_gap × 0.30
--       accuracy_gap   × 0.30
--       recency_penalty × 0.15
--     ) × urgency × (priority/3)
--   Top-scored chapters → chapter missions
--   If no recent paper → add paper mission
--   Maximum 3 missions per day

CREATE OR REPLACE FUNCTION public.generate_daily_missions(p_user_id UUID)
RETURNS INTEGER  -- returns number of missions generated
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_today          DATE := CURRENT_DATE;
  v_existing_count INTEGER;
  v_inserted_count INTEGER := 0;
  v_budget         INTEGER;
  v_max_missions   INTEGER;
  rec              RECORD;
  v_mission_type   mission_type_enum;
  v_title          TEXT;
  v_description    TEXT;
  v_xp_reward      SMALLINT;
BEGIN
  -- Get user's max missions preference (default 3)
  SELECT COALESCE(max_missions_per_day, 3) INTO v_max_missions
  FROM public.user_settings
  WHERE user_id = p_user_id;

  -- How many missions already exist today?
  SELECT COUNT(*) INTO v_existing_count
  FROM public.daily_missions
  WHERE user_id = p_user_id AND mission_date = v_today;

  v_budget := v_max_missions - v_existing_count;
  IF v_budget <= 0 THEN
    RETURN 0;
  END IF;

  -- ── CHAPTER MISSIONS: top-scored chapters ─────────────────────────────
  FOR rec IN
    WITH chapter_scores AS (
      SELECT
        uc.id                     AS user_chapter_id,
        uc.chapter_id,
        c.title                   AS chapter_title,
        s.id                      AS subject_id,
        s.name                    AS subject_name,
        us.exam_date,
        us.priority,
        uc.notes_status,
        uc.confidence_level,
        uc.last_reviewed_at,

        -- Notes gap: 1 if not complete, 0 if complete
        CASE WHEN uc.notes_status != 'complete' THEN 1.0 ELSE 0.0 END
          AS notes_gap,

        -- Confidence gap: (5 - level) / 4, default to 3/5 if not set
        (5.0 - COALESCE(uc.confidence_level, 3)) / 4.0
          AS confidence_gap,

        -- Accuracy gap: 1 - avg(marks_obtained/marks_available) for this chapter
        -- Defaults to 0.3 (assumed 70% accuracy) if no data
        (1.0 - COALESCE(
          (
            SELECT AVG(pqa.marks_obtained::NUMERIC / NULLIF(pqa.marks_available, 0))
            FROM public.paper_question_attempts pqa
            JOIN public.past_papers pp ON pp.id = pqa.paper_id
            WHERE pqa.chapter_id = uc.chapter_id
              AND pp.user_id = p_user_id
          ),
          0.7
        )) AS accuracy_gap,

        -- Recency penalty: days since last reviewed, capped at 14
        LEAST(
          COALESCE(
            EXTRACT(DAY FROM (NOW() - uc.last_reviewed_at))::NUMERIC,
            14  -- never reviewed = max penalty
          ),
          14
        ) / 14.0 AS recency_penalty,

        -- Urgency: 1/days_until_exam (higher urgency = sooner exam)
        -- Clamped to avoid division by zero; cap at 0.1 so far-away exams still matter
        GREATEST(
          1.0 / NULLIF(
            EXTRACT(DAY FROM (us.exam_date - CURRENT_DATE))::NUMERIC, 0
          ),
          0.01
        ) AS urgency

      FROM public.user_chapters uc
      JOIN public.chapters c ON c.id = uc.chapter_id
      JOIN public.subjects s ON s.id = c.subject_id
      JOIN public.user_subjects us ON us.user_id = p_user_id
        AND us.subject_id = s.id
      WHERE uc.user_id = p_user_id
        AND us.exam_date IS NOT NULL
        AND us.exam_date > CURRENT_DATE
        AND us.is_archived = FALSE
    )
    SELECT
      *,
      (
        notes_gap      * 0.25 +
        confidence_gap * 0.30 +
        accuracy_gap   * 0.30 +
        recency_penalty * 0.15
      ) * urgency * (priority::NUMERIC / 3.0)  AS chapter_score
    FROM chapter_scores
    ORDER BY chapter_score DESC
    LIMIT v_budget
  LOOP
    -- Determine mission type based on primary gap
    IF rec.notes_status = 'none' THEN
      v_mission_type  := 'complete_notes';
      v_title         := 'Complete Chapter Notes';
      v_description   := rec.chapter_title || ' · ' || rec.subject_name;
      v_xp_reward     := 50;
    ELSIF rec.accuracy_gap > 0.4 THEN
      v_mission_type  := 'revisit_weak_topic';
      v_title         := 'Revisit Weak Topic';
      v_description   := rec.chapter_title || ' — low accuracy detected';
      v_xp_reward     := 40;
    ELSIF rec.notes_status = 'in_progress' THEN
      v_mission_type  := 'complete_notes';
      v_title         := 'Finish Chapter Notes';
      v_description   := rec.chapter_title || ' · ' || rec.subject_name;
      v_xp_reward     := 50;
    ELSE
      v_mission_type  := 'review_chapter';
      v_title         := 'Review Chapter';
      v_description   := rec.chapter_title || ' · ' || rec.subject_name;
      v_xp_reward     := 30;
    END IF;

    INSERT INTO public.daily_missions (
      user_id, mission_date, type, target_entity_type,
      target_entity_id, title, description, xp_reward, status
    )
    VALUES (
      p_user_id, v_today, v_mission_type, 'chapter',
      rec.user_chapter_id, v_title, v_description, v_xp_reward, 'pending'
    )
    ON CONFLICT (user_id, mission_date, type, target_entity_id) DO NOTHING;

    GET DIAGNOSTICS v_inserted_count = v_inserted_count + ROW_COUNT;
    v_budget := v_budget - 1;
    EXIT WHEN v_budget <= 0;
  END LOOP;

  -- ── PAPER MISSION: if no paper in last 7 days ────────────────────────
  IF v_budget > 0 THEN
    IF NOT EXISTS (
      SELECT 1 FROM public.past_papers pp
      WHERE pp.user_id = p_user_id
        AND pp.created_at > NOW() - INTERVAL '7 days'
    ) THEN
      INSERT INTO public.daily_missions (
        user_id, mission_date, type, target_entity_type,
        target_entity_id, title, description, xp_reward, status
      )
      SELECT
        p_user_id, v_today, 'attempt_paper', 'subject',
        us.subject_id,
        'Attempt a Past Paper',
        'Practice with a ' || s.name || ' past paper',
        75,
        'pending'
      FROM public.user_subjects us
      JOIN public.subjects s ON s.id = us.subject_id
      WHERE us.user_id = p_user_id
        AND us.exam_date IS NOT NULL
        AND us.exam_date > CURRENT_DATE
        AND us.is_archived = FALSE
      ORDER BY us.exam_date ASC  -- most urgent exam first
      LIMIT 1
      ON CONFLICT (user_id, mission_date, type, target_entity_id) DO NOTHING;

      GET DIAGNOSTICS v_inserted_count = v_inserted_count + ROW_COUNT;
    END IF;
  END IF;

  RETURN v_inserted_count;
END;
$$;

COMMENT ON FUNCTION public.generate_daily_missions IS
  'Mission Engine: generates up to max_missions_per_day missions based on weighted chapter scores. '
  'Weights: notes_gap 25%, confidence_gap 30%, accuracy_gap 30%, recency_penalty 15%. '
  'Multiplied by urgency (1/days_to_exam) and subject priority. '
  'ON CONFLICT ensures idempotency — safe to call multiple times.';


-- ─── COMPLETE_MISSION ──────────────────────────────────────────────────────
-- Atomic RPC: marks mission done, awards XP, updates streak, checks achievements.
-- Called by the client via supabase.rpc('complete_mission', {mission_id}).
-- Returns enriched response for the client to show XP and achievement toasts.

CREATE OR REPLACE FUNCTION public.complete_mission(
  p_mission_id UUID,
  p_user_id    UUID  -- explicit for RLS verification
)
RETURNS JSONB  -- { xp_awarded, new_total_xp, streak_days, achievements_unlocked[] }
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_mission        public.daily_missions%ROWTYPE;
  v_profile        public.profiles%ROWTYPE;
  v_streak         public.streaks%ROWTYPE;
  v_achievements   JSONB := '[]'::JSONB;
  v_achievement_xp SMALLINT := 0;
  v_result         JSONB;
  rec              RECORD;
BEGIN
  -- Verify ownership and fetch mission
  SELECT * INTO v_mission
  FROM public.daily_missions
  WHERE id = p_mission_id AND user_id = p_user_id AND status = 'pending'
  FOR UPDATE;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Mission not found or already completed: %', p_mission_id
      USING ERRCODE = 'P0001';
  END IF;

  -- Mark mission as completed
  UPDATE public.daily_missions
  SET status       = 'completed',
      completed_at = NOW()
  WHERE id = p_mission_id;

  -- Award mission XP
  PERFORM public.award_xp(
    p_user_id,
    v_mission.xp_reward,
    'mission_complete',
    p_mission_id,
    jsonb_build_object('mission_type', v_mission.type)
  );

  -- If all missions for today are done, award completion bonus
  IF NOT EXISTS (
    SELECT 1 FROM public.daily_missions
    WHERE user_id = p_user_id
      AND mission_date = CURRENT_DATE
      AND status = 'pending'
      AND id != p_mission_id
  ) THEN
    PERFORM public.award_xp(
      p_user_id, 25, 'mission_complete',
      NULL, '{"bonus": "all_missions_complete"}'::JSONB
    );
  END IF;

  -- Update streak
  PERFORM public.update_streak(p_user_id);

  -- Update last_reviewed_at on the target chapter (if chapter mission)
  IF v_mission.target_entity_type = 'chapter' AND v_mission.target_entity_id IS NOT NULL THEN
    UPDATE public.user_chapters
    SET last_reviewed_at = NOW(), updated_at = NOW()
    WHERE id = v_mission.target_entity_id AND user_id = p_user_id;
  END IF;

  -- Check and unlock achievements, collect results
  FOR rec IN
    SELECT * FROM public.check_and_unlock_achievements(p_user_id)
  LOOP
    v_achievements := v_achievements || jsonb_build_object(
      'key', rec.achievement_key,
      'xp', rec.xp_reward
    );
    v_achievement_xp := v_achievement_xp + rec.xp_reward;
  END LOOP;

  -- Fetch updated profile for response
  SELECT * INTO v_profile FROM public.profiles WHERE id = p_user_id;
  SELECT * INTO v_streak  FROM public.streaks  WHERE user_id = p_user_id;

  -- Build response payload
  v_result := jsonb_build_object(
    'xp_awarded',           v_mission.xp_reward,
    'achievement_xp',       v_achievement_xp,
    'new_total_xp',         v_profile.total_xp,
    'new_level',            v_profile.current_level,
    'level_title',          public.compute_level_title(v_profile.current_level),
    'streak_days',          v_streak.current_streak,
    'achievements_unlocked', v_achievements
  );

  RETURN v_result;
END;
$$;

COMMENT ON FUNCTION public.complete_mission IS
  'Atomic mission completion: marks done, awards XP, updates streak, checks achievements. '
  'Returns enriched JSONB for client toast notifications. '
  'Use: supabase.rpc("complete_mission", { mission_id, p_user_id })';


-- ─── COMPUTE_READINESS_SCORE ───────────────────────────────────────────────
-- Readiness Score formula (per system design):
--   score = (notes_pct × 0.35) + (avg_paper_accuracy × 0.40) + (avg_confidence × 0.25)
-- Returns 0–100 as NUMERIC(5,2).
-- Overloaded: pass null subject_id for overall score across all subjects.

CREATE OR REPLACE FUNCTION public.compute_readiness_score(
  p_user_id    UUID,
  p_subject_id UUID DEFAULT NULL  -- NULL = overall across all subjects
)
RETURNS NUMERIC(5,2)
LANGUAGE plpgsql
STABLE
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_notes_pct          NUMERIC;
  v_avg_paper_accuracy NUMERIC;
  v_avg_confidence     NUMERIC;
BEGIN
  -- ── Notes completion percentage (weight: 35%) ──────────────────────────
  SELECT
    CASE
      WHEN COUNT(*) = 0 THEN 0
      ELSE COUNT(*) FILTER (WHERE uc.notes_status = 'complete')::NUMERIC / COUNT(*)
    END
  INTO v_notes_pct
  FROM public.user_chapters uc
  JOIN public.chapters c ON c.id = uc.chapter_id
  WHERE uc.user_id = p_user_id
    AND (p_subject_id IS NULL OR c.subject_id = p_subject_id);

  -- ── Average paper accuracy (weight: 40%) ──────────────────────────────
  SELECT COALESCE(AVG(pp.accuracy_pct) / 100.0, 0)
  INTO v_avg_paper_accuracy
  FROM public.past_papers pp
  WHERE pp.user_id = p_user_id
    AND (p_subject_id IS NULL OR pp.subject_id = p_subject_id);

  -- ── Average confidence normalised to 0–1 (weight: 25%) ────────────────
  SELECT COALESCE(AVG(uc.confidence_level) / 5.0, 0)
  INTO v_avg_confidence
  FROM public.user_chapters uc
  JOIN public.chapters c ON c.id = uc.chapter_id
  WHERE uc.user_id = p_user_id
    AND uc.confidence_level IS NOT NULL
    AND (p_subject_id IS NULL OR c.subject_id = p_subject_id);

  -- Weighted formula
  RETURN ROUND(
    (v_notes_pct * 0.35 + v_avg_paper_accuracy * 0.40 + v_avg_confidence * 0.25) * 100,
    2
  );
END;
$$;

COMMENT ON FUNCTION public.compute_readiness_score IS
  'Readiness Score = notes_pct(35%) + paper_accuracy(40%) + confidence(25%). '
  'Pass NULL subject_id for overall score. Returns 0.00–100.00.';


-- ─── GET_USER_DASHBOARD_STATS ──────────────────────────────────────────────
-- Single RPC to fetch all dashboard data in one round-trip.
-- Avoids N+1 queries from the client fetching each widget separately.

CREATE OR REPLACE FUNCTION public.get_user_dashboard_stats(p_user_id UUID)
RETURNS JSONB
LANGUAGE plpgsql
STABLE
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_profile    public.profiles%ROWTYPE;
  v_streak     public.streaks%ROWTYPE;
  v_readiness  NUMERIC(5,2);
  v_result     JSONB;
BEGIN
  -- Auth check
  IF auth.uid() != p_user_id THEN
    RAISE EXCEPTION 'Unauthorized' USING ERRCODE = '42501';
  END IF;

  SELECT * INTO v_profile FROM public.profiles WHERE id = p_user_id;
  SELECT * INTO v_streak  FROM public.streaks  WHERE user_id = p_user_id;
  v_readiness := public.compute_readiness_score(p_user_id, NULL);

  SELECT jsonb_build_object(
    'profile', jsonb_build_object(
      'full_name',      v_profile.full_name,
      'avatar_url',     v_profile.avatar_url,
      'total_xp',       v_profile.total_xp,
      'current_level',  v_profile.current_level,
      'level_title',    public.compute_level_title(v_profile.current_level)
    ),
    'streak', jsonb_build_object(
      'current',    v_streak.current_streak,
      'longest',    v_streak.longest_streak,
      'last_date',  v_streak.last_activity_date
    ),
    'overall_readiness',  v_readiness,
    'today_missions', (
      SELECT json_agg(row_to_json(dm.*))
      FROM public.daily_missions dm
      WHERE dm.user_id = p_user_id
        AND dm.mission_date = CURRENT_DATE
    ),
    'subject_readiness', (
      SELECT json_agg(jsonb_build_object(
        'subject_id',   us.subject_id,
        'subject_name', s.name,
        'color_hex',    s.color_hex,
        'exam_date',    us.exam_date,
        'days_until',   (us.exam_date - CURRENT_DATE),
        'readiness',    public.compute_readiness_score(p_user_id, us.subject_id)
      ))
      FROM public.user_subjects us
      JOIN public.subjects s ON s.id = us.subject_id
      WHERE us.user_id = p_user_id AND us.is_archived = FALSE
    ),
    'recent_xp_events', (
      SELECT json_agg(row_to_json(xe.*))
      FROM (
        SELECT * FROM public.xp_events
        WHERE user_id = p_user_id
        ORDER BY created_at DESC
        LIMIT 10
      ) xe
    )
  ) INTO v_result;

  RETURN v_result;
END;
$$;


-- ─── GET_LEADERBOARD ───────────────────────────────────────────────────────
-- Returns XP leaderboard. scope: 'all_time' | 'weekly'.
-- For friends leaderboard (future): pass p_friend_ids array.

CREATE OR REPLACE FUNCTION public.get_leaderboard(
  p_scope      TEXT    DEFAULT 'all_time',   -- 'all_time' | 'weekly'
  p_limit      INTEGER DEFAULT 50
)
RETURNS TABLE(
  rank          BIGINT,
  user_id       UUID,
  full_name     TEXT,
  avatar_url    TEXT,
  current_level SMALLINT,
  level_title   TEXT,
  xp_value      BIGINT,
  streak_days   INTEGER
)
LANGUAGE plpgsql
STABLE
SECURITY DEFINER
SET search_path = public
AS $$
BEGIN
  IF p_scope = 'weekly' THEN
    RETURN QUERY
      SELECT
        ROW_NUMBER() OVER (ORDER BY SUM(xe.xp_amount) DESC) AS rank,
        p.id,
        p.full_name,
        p.avatar_url,
        p.current_level,
        public.compute_level_title(p.current_level),
        SUM(xe.xp_amount)::BIGINT,
        s.current_streak
      FROM public.profiles p
      LEFT JOIN public.xp_events xe ON xe.user_id = p.id
        AND xe.created_at >= DATE_TRUNC('week', NOW())
      LEFT JOIN public.streaks s ON s.user_id = p.id
      GROUP BY p.id, p.full_name, p.avatar_url, p.current_level, s.current_streak
      ORDER BY SUM(xe.xp_amount) DESC NULLS LAST
      LIMIT p_limit;
  ELSE
    RETURN QUERY
      SELECT
        ROW_NUMBER() OVER (ORDER BY p.total_xp DESC) AS rank,
        p.id,
        p.full_name,
        p.avatar_url,
        p.current_level,
        public.compute_level_title(p.current_level),
        p.total_xp::BIGINT,
        s.current_streak
      FROM public.profiles p
      LEFT JOIN public.streaks s ON s.user_id = p.id
      ORDER BY p.total_xp DESC
      LIMIT p_limit;
  END IF;
END;
$$;
```

```sql
-- ============================================================
-- FILE: supabase/migrations/009b_achievement_function.sql
-- PURPOSE: Achievement checking function (separate file due to
-- dependency on achievement_definitions table + other tables).
-- ============================================================

CREATE OR REPLACE FUNCTION public.check_and_unlock_achievements(p_user_id UUID)
RETURNS TABLE(achievement_key TEXT, xp_reward SMALLINT)
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_rec             RECORD;
  v_condition_met   BOOLEAN;
BEGIN
  -- Iterate over active achievements not yet unlocked by this user
  FOR v_rec IN
    SELECT ad.*
    FROM public.achievement_definitions ad
    WHERE ad.is_active = TRUE
      AND NOT EXISTS (
        SELECT 1 FROM public.user_achievements ua
        WHERE ua.user_id = p_user_id
          AND ua.achievement_key = ad.key
      )
  LOOP
    v_condition_met := FALSE;

    CASE v_rec.key

      WHEN 'first_blood' THEN
        -- First chapter notes completed
        SELECT COUNT(*) > 0 INTO v_condition_met
        FROM public.user_chapters uc
        WHERE uc.user_id = p_user_id AND uc.notes_status = 'complete';

      WHEN 'paper_hunter' THEN
        -- First past paper logged
        SELECT COUNT(*) > 0 INTO v_condition_met
        FROM public.past_papers pp WHERE pp.user_id = p_user_id;

      WHEN 'five_papers' THEN
        SELECT COUNT(*) >= 5 INTO v_condition_met
        FROM public.past_papers pp WHERE pp.user_id = p_user_id;

      WHEN 'ten_papers' THEN
        SELECT COUNT(*) >= 10 INTO v_condition_met
        FROM public.past_papers pp WHERE pp.user_id = p_user_id;

      WHEN 'ace' THEN
        -- 90%+ on any paper
        SELECT COUNT(*) > 0 INTO v_condition_met
        FROM public.past_papers pp
        WHERE pp.user_id = p_user_id AND pp.accuracy_pct >= 90;

      WHEN 'perfect_score' THEN
        -- 100% on any paper
        SELECT COUNT(*) > 0 INTO v_condition_met
        FROM public.past_papers pp
        WHERE pp.user_id = p_user_id AND pp.accuracy_pct = 100;

      WHEN 'streak_7' THEN
        SELECT current_streak >= 7 INTO v_condition_met
        FROM public.streaks WHERE user_id = p_user_id;

      WHEN 'streak_30' THEN
        SELECT current_streak >= 30 INTO v_condition_met
        FROM public.streaks WHERE user_id = p_user_id;

      WHEN 'streak_100' THEN
        SELECT current_streak >= 100 INTO v_condition_met
        FROM public.streaks WHERE user_id = p_user_id;

      WHEN 'completionist' THEN
        -- All global chapters for at least one enrolled subject are complete
        SELECT EXISTS (
          SELECT us.subject_id
          FROM public.user_subjects us
          WHERE us.user_id = p_user_id AND us.is_archived = FALSE
          GROUP BY us.subject_id
          HAVING
            (SELECT COUNT(*) FROM public.chapters c
             WHERE c.subject_id = us.subject_id AND c.is_global = TRUE)
            =
            (SELECT COUNT(*) FROM public.user_chapters uc
             JOIN public.chapters c ON c.id = uc.chapter_id
             WHERE uc.user_id = p_user_id
               AND c.subject_id = us.subject_id
               AND uc.notes_status = 'complete')
        ) INTO v_condition_met;

      WHEN 'speed_run' THEN
        -- 3 missions completed in one day
        SELECT COUNT(*) >= 3 INTO v_condition_met
        FROM public.daily_missions dm
        WHERE dm.user_id = p_user_id
          AND dm.status = 'completed'
          AND dm.mission_date = CURRENT_DATE;

      WHEN 'night_owl' THEN
        -- Study activity between midnight and 4am
        SELECT COUNT(*) > 0 INTO v_condition_met
        FROM public.xp_events xe
        WHERE xe.user_id = p_user_id
          AND EXTRACT(HOUR FROM xe.created_at) < 4;

      WHEN 'multi_subject' THEN
        -- Enrolled in 3+ subjects
        SELECT COUNT(*) >= 3 INTO v_condition_met
        FROM public.user_subjects us
        WHERE us.user_id = p_user_id AND us.is_archived = FALSE;

      WHEN 'level_5' THEN
        SELECT current_level >= 5 INTO v_condition_met
        FROM public.profiles WHERE id = p_user_id;

      WHEN 'level_10' THEN
        SELECT current_level >= 10 INTO v_condition_met
        FROM public.profiles WHERE id = p_user_id;

      WHEN 'atlas_legend' THEN
        SELECT current_level >= 15 INTO v_condition_met
        FROM public.profiles WHERE id = p_user_id;

      WHEN 'consistent' THEN
        -- Completed missions on 14 distinct days
        SELECT COUNT(DISTINCT mission_date) >= 14 INTO v_condition_met
        FROM public.daily_missions dm
        WHERE dm.user_id = p_user_id AND dm.status = 'completed';

      WHEN 'high_confidence' THEN
        -- Average confidence >= 4 across all chapters
        SELECT AVG(confidence_level) >= 4 INTO v_condition_met
        FROM public.user_chapters uc
        WHERE uc.user_id = p_user_id AND uc.confidence_level IS NOT NULL;

      WHEN 'docs_connected' THEN
        -- Connected Google Docs
        SELECT COUNT(*) > 0 INTO v_condition_met
        FROM public.google_docs_tokens WHERE user_id = p_user_id;

      WHEN 'linked_notes' THEN
        -- Linked a Google Doc to at least 5 chapters
        SELECT COUNT(*) >= 5 INTO v_condition_met
        FROM public.user_chapters uc
        WHERE uc.user_id = p_user_id AND uc.google_doc_id IS NOT NULL;

      ELSE
        -- Unknown achievement key: skip silently
        v_condition_met := FALSE;
    END CASE;

    IF v_condition_met THEN
      -- Insert achievement (ON CONFLICT in case of race condition)
      INSERT INTO public.user_achievements (user_id, achievement_key)
      VALUES (p_user_id, v_rec.key)
      ON CONFLICT (user_id, achievement_key) DO NOTHING;

      -- Award XP for the achievement (only if actually inserted)
      IF FOUND AND v_rec.xp_reward > 0 THEN
        PERFORM public.award_xp(
          p_user_id,
          v_rec.xp_reward,
          'achievement_unlock',
          NULL,
          jsonb_build_object('achievement_key', v_rec.key)
        );
      END IF;

      achievement_key := v_rec.key;
      xp_reward       := v_rec.xp_reward;
      RETURN NEXT;
    END IF;
  END LOOP;
END;
$$;
```

## Design Notes & Rationale

### Why each architectural decision was made

| Decision | Rationale |
|---|---|
| **`accuracy_pct` as GENERATED column** | Zero chance of drift between stored value and actual score. No trigger required. |
| **`profiles.total_xp` denormalised** | Avoids `SUM(xp_events)` on every dashboard load. Kept in sync by `sync_xp_to_profile` trigger. |
| **XP events as immutable ledger** | Enables auditing, leaderboard recalculation, and future refund mechanics without data loss. |
| **`FOR UPDATE` lock in `update_streak`** | Prevents race condition where two simultaneous mission completions both increment the streak by 1. |
| **`SECURITY DEFINER` on all engine functions** | Allows server functions to bypass RLS while clients remain fully isolated. Search path set explicitly to prevent injection. |
| **`ON CONFLICT DO NOTHING` in mission generation** | Makes `generate_daily_missions` safe to call multiple times (idempotent). |
| **`user_id_1 < user_id_2` constraint on friendships** | One row per pair. Prevents duplicates like (A,B) and (B,A) without app-layer logic. |
| **`achievement_definitions` in DB** | Allows adding/modifying badges without code deploys. Admin can toggle `is_active`. |
| **Partial indexes** | `WHERE status = 'pending'`, `WHERE is_read = FALSE` etc. reduce index size and speed up the most common filtered queries. |
| **Future tables as stubs now** | Schema migrations are painful with production data. Defining FKs and indexes early avoids multi-step migrations later. |
| **`study_pets`, `pvp_challenges`, `shop_items`** | All have `is_active`/`is_available` guards so they can exist in DB without the application exposing them yet. |
| **`google_docs_tokens` — zero RLS policies** | With RLS enabled but no policies, the table is completely inaccessible to all clients. Only `SECURITY DEFINER` server functions can read it. |
